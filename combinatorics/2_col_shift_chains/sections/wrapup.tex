  \begin{center}
    \only<1>{
      \begin{shiftchain}[
        column sep=6mm, row sep=5mm, dot radius=3.0pt,
        label font=\scriptsize,
        bars={4}, bar labels={4/$x_i$},
      ]{12}
        \shiftedge{2/cfred,5/cfnavy,8/cfnavy,12/cfnavy}[$\vec f$]
      \end{shiftchain}
    }
    \only<2->{
      \begin{shiftchain}[
        column sep=6mm, row sep=5mm, dot radius=3.0pt,
        label font=\scriptsize,
        bars={4,10}, bar labels={4/$x_i$,10/$x_{i+1}$},
      ]{12}
        \shiftedge{2/cfred,5/cfnavy,8/cfnavy,12/cfgr}[$\vec f$]
      \end{shiftchain}
    }
  \end{center}

  \begin{itemize}
    \item<1-> X1 gives $|S_{c_0}(\vec f)|\in\{1,2\}$ for every edge.
    \item<2-> Start with colour $0$ and advance
      $0\to1\to2\to0$ at each switch.
    \item<3-> $c_3(f_4)-c_3(f_1)\equiv
      |S_{c_0}(\vec f)|\pmod 3$, and
      $|S_{c_0}(\vec f)|\in\{1,2\}$.
      Thus no edge is monochromatic.
  \end{itemize}
\end{frame}
